1. Transactional Email Engine

The kora_runtime::email module provides an asynchronous email delivery engine supporting TLS, authentication, HTML templates, and binary attachments.

Sending Verification Email 
use kora_runtime::email::{EmailService, EmailMessage, SmtpConfig};

let config = SmtpConfig::new("smtp.sendgrid.net", 587)
    .with_credentials("apikey", "SG.secret_key_token")
    .with_tls(true);

let service = EmailService::new(config);

// Generate automated verification email with token
let verification_email = service.create_verification_email(
    "user@example.com",
    "Sara Smith",
    "https://api.kora.dev/v1/auth/verify?token=secure_tok_123"
);

match service.send(&verification_email) {
    Ok(msg_id) => println!("Email sent successfully! Message ID: {}", msg_id),
    Err(e) => eprintln!("Failed to deliver email: {}", e),
}

2. Image Processing & Media Security Pipeline

Handling user file uploads without validation is a major security vulnerability. Kora's media engine protects against Decompression Bombs (Pixel Floods), SVG Stored XSS, and EXIF Geolocation Privacy Leaks:

Security Validation, EXIF Stripping & Resizing 
use kora_runtime::media::{ImageProcessor, ImageFormat, UploadSecurityValidator};

// 1. Validate inbound file security (Max 5MB, JPG/PNG/WebP/SVG, Max 50M Pixels)
let validator = UploadSecurityValidator::new(5 * 1024 * 1024)
    .allow_format(ImageFormat::Jpeg)
    .allow_format(ImageFormat::Png)
    .allow_format(ImageFormat::Webp)
    .allow_format(ImageFormat::Svg)
    .with_max_pixels(50_000_000);

let format = validator.validate(&file_bytes, "avatar.png")?;

// 2. Load into processor & strip privacy-sensitive EXIF camera/GPS tags
let mut processor = ImageProcessor::from_bytes(file_bytes, 1920, 1080)?;
let bytes_removed = processor.strip_metadata();
println!("Stripped {} bytes of EXIF metadata", bytes_removed);

// 3. Proportional resize maintaining aspect ratio
let (target_w, target_h) = processor.resize(800, 600, true);
println!("Resized dimensions: {}x{}", target_w, target_h); // 800x450

// 4. Generate thumbnail
let (thumb_w, thumb_h) = processor.thumbnail(150);
println!("Thumbnail dimensions: {}x{}", thumb_w, thumb_h);

3. PDF Generation, Layout Positioning & Commercial Invoices

Design beautiful documents with fine-grained control over page dimensions, margins, typography sizes, colors, borders, diagonal watermarks, and access permissions.

Designing Styled PDF Document 
use kora_runtime::document::pdf::{
    PdfDocument, PageSize, Margins, PdfTable, PdfColor, TableCell, PdfSecurity
};

let mut doc = PdfDocument::new("Quarterly Enterprise Audit")
    .with_page_size(PageSize::A4)
    .with_margins(Margins::uniform(45.0));

// Set document header and running footer
doc.set_header("Hwai Technology Group — Confidential Report");
doc.set_footer("Page 1 of 1 — Generated by Kora Platform");

// Add styled typography
doc.add_heading("Financial Summary & Compliance", 1);
doc.add_paragraph("All services verified with 99.99% availability.");
doc.add_spacer(12.0);

// Add table with zebra striping and custom headers
let mut table = PdfTable::new();
table.set_zebra_striping(true, PdfColor::LIGHT_GRAY);
table.add_header_row(&["Service Cluster", "P99 Latency", "Monthly Cost"]);
table.add_data_row(&["Authentication", "1.4ms", "$1,250.00"]);
table.add_data_row(&["Payment Gateway", "8.1ms", "$4,800.00"]);
doc.add_table(table);

// Add diagonal watermark and password protection
doc.add_watermark("CONFIDENTIAL", 0.15);
doc.set_security(PdfSecurity {
    owner_password: Some("admin_secret".into()),
    user_password: None,
    can_print: true,
    can_copy_text: false,
    can_modify: false,
    can_annotate: true,
});

let pdf_bytes = doc.render_to_bytes();
Building Commercial Invoice with Tax & Discounts 
use kora_runtime::document::pdf::InvoicePdfBuilder;

let mut builder = InvoicePdfBuilder::new("INV-2026-0089", "Acme Corporation")
    .with_company_info("Kora Global Inc.", Some("US-EIN-987654321".into()))
    .with_currency("$", "USD")
    .with_tax_percent(15)
    .with_global_discount(20000) // $200.00 discount
    .with_dates("2026-11-17", Some("2026-12-17".into()));

builder.add_line_item("Dedicated Cloud Cluster (48 Cores)", 2, 150000); // 2 x $1,500.00
builder.add_line_item("Managed Redis Enterprise Cache", 1, 45000);     // 1 x $450.00

println!("Subtotal:    {}", builder.format_money(builder.calculate_subtotal()));
println!("Discount:   -{}", builder.format_money(builder.calculate_discount()));
println!("VAT (15%):   {}", builder.format_money(builder.calculate_tax()));
println!("Total Due:   {}", builder.format_money(builder.calculate_total()));

let invoice_doc = builder.build();
let invoice_pdf = invoice_doc.render_to_bytes();

4. Excel Workbooks & Formatted Spreadsheets

Export multi-sheet workbooks with currency formatting, auto-fitting column widths, and formula summations:

Formatted Worksheet with Summation 
use kora_runtime::document::excel::{Worksheet, Workbook, CellValue, CellStyle, NumberFormat};

let mut sheet = Worksheet::new("Q4 Revenue");

// Styled Header Row
sheet.set_cell_styled(1, 1, CellValue::Text("Product Name".into()), CellStyle::bold_header());
sheet.set_cell_styled(1, 2, CellValue::Text("Quarter".into()), CellStyle::bold_header());
sheet.set_cell_styled(1, 3, CellValue::Text("Revenue".into()), CellStyle::bold_header());

// Data Rows with Currency Formatting
let currency_style = CellStyle::currency('$');
sheet.set_cell(2, 1, CellValue::Text("Enterprise Subscription".into()));
sheet.set_cell(2, 2, CellValue::Text("Q4".into()));
sheet.set_cell_styled(2, 3, CellValue::Number(125000.0), currency_style.clone());

sheet.set_cell(3, 1, CellValue::Text("Professional Support Plan".into()));
sheet.set_cell(3, 2, CellValue::Text("Q4".into()));
sheet.set_cell_styled(3, 3, CellValue::Number(45000.0), currency_style);

// Calculate Totals & Auto-Fit Column Widths
let total_revenue = sheet.calculate_sum(3, 2, 3);
let avg_revenue = sheet.calculate_average(3, 2, 3).unwrap();
sheet.auto_fit_columns();

println!("Total Revenue: ${:.2}, Average: ${:.2}", total_revenue, avg_revenue);

// Export to CSV or TSV
let csv = sheet.export_to_csv();
let tsv = sheet.export_to_tsv();

5. Secure Cloud Object Storage (S3 / Local / Memory)

Enterprise storage with Path Traversal Defense (CWE-22), storage quota management, and pre-signed URLs:

Safe Storage with Path Traversal Defense 
use kora_runtime::storage::{StorageManager, MemoryStorageDriver, sanitize_storage_key};

// Reject malicious path traversal keys automatically
assert!(sanitize_storage_key("../../../etc/passwd").is_err());
assert!(sanitize_storage_key("safe/bucket/invoices/inv_101.pdf").is_ok());

let driver = MemoryStorageDriver::new("https://cdn.kora.dev")
    .with_quota(10 * 1024 * 1024 * 1024); // 10 GB Quota

let storage = StorageManager::new(driver);

// Store file with SHA-256 checksum verification
let metadata = storage.put("invoices/2026/inv_101.pdf", pdf_bytes, "application/pdf")?;
println!("Stored key: {}, Checksum: {}", metadata.key, metadata.checksum);

// Generate pre-signed download URL valid for 2 hours
let presigned_url = storage.generate_presigned_url("invoices/2026/inv_101.pdf", 7200)?;
println!("Secure URL: {}", presigned_url);

// List files under a prefix
let files = storage.list_prefix("invoices/2026/", 50)?;
println!("Found {} files", files.len());
Unified Storage API 
use kora_runtime::storage::{StorageManager, MemoryStorageDriver};

// Inject memory driver for tests or S3 driver for production
let storage = StorageManager::new(MemoryStorageDriver::new());

// Put object
storage.put("reports/q1_2026.pdf", pdf_bytes, "application/pdf").await?;

// Generate secure pre-signed download URL valid for 1 hour
let download_url = storage.generate_presigned_url("reports/q1_2026.pdf", 3600)?;
println!("Presigned URL: {}", download_url);

6. Multi-Format Reporting Engine

Define a report schema once and export it on demand to CSV, HTML, or PDF formats:

Exporting Multi-Format Report 
use kora_runtime::document::reporting::{ReportDataset, ReportEngine, ReportFormat};

let dataset = ReportDataset {
    title: "Quarterly Performance Audit".into(),
    headers: vec!["Service".into(), "P99 Latency".into(), "Availability".into()],
    rows: vec![
        vec!["Auth Service".into(), "1.2ms".into(), "99.99%".into()],
        vec!["Payment Gateway".into(), "8.4ms".into(), "99.95%".into()],
        vec!["Orders Worker".into(), "3.1ms".into(), "99.98%".into()],
    ],
};

let html_report = ReportEngine::generate(&dataset, ReportFormat::Html)?;
let csv_report  = ReportEngine::generate(&dataset, ReportFormat::Csv)?;
let pdf_report  = ReportEngine::generate(&dataset, ReportFormat::Pdf)?;
